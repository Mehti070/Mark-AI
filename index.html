"model" : "user", parts: [{ text: m.content }], })), }), } ); if (!r.ok) throw new Error("gemini " + r.status + " " + (await r.text()).slice(0, 200)); const d = await r.json(); const text = (d.candidates?.[0]?.content?.parts || []).map((p) => p.text || "").join(""); if (!text) throw new Error("gemini empty"); return text; } async function askClaude(env, system, messages) { if (!env.ANTHROPIC_API_KEY) throw new Error("no claude key"); const r = await fetch("https://api.anthropic.com/v1/messages", { method: "POST", headers: { "Content-Type": "application/json", "x-api-key": env.ANTHROPIC_API_KEY, "anthropic-version": "2023-06-01", }, body: JSON.stringify({ model: env.CLAUDE_MODEL || "claude-sonnet-5-5", max_tokens: 2048, system, messages: messages.map((m) => ({ role: m.role, content: m.content })), }), }); if (!r.ok) throw new Error("claude " + r.status + " " + (await r.text()).slice(0, 200)); const d = await r.json(); const text = (d.content || []).map((p) => p.text || "").join(""); if (!text) throw new Error("claude empty"); return text; } export default { async fetch(req, env) { const h = cors(env); if (req.method === "OPTIONS") return new Response(null, { headers: h }); if (req.method !== "POST") return new Response("Mark AI server", { headers: h }); if (env.ACCESS_TOKEN && req.headers.get("X-Access") !== env.ACCESS_TOKEN) return new Response("Неверный ключ доступа", { status: 401, headers: h }); let body; try { body = await req.json(); } catch { return new Response("Bad request", { status: 400, headers: h }); } const system = body.system || "Ты Mark AI — личный ИИ-помощник."; const messages = Array.isArray(body.messages) ? body.messages.slice(-30) : []; if (!messages.length) return new Response("Пустой запрос", { status: 400, headers: h }); // Спрашиваем обе модели одновременно const [g, c] = await Promise.allSettled([ askGemini(env, system, messages), askClaude(env, system, messages), ]); const drafts = [g, c].filter((x) => x.status === "fulfilled").map((x) => x.value); let answer; if (drafts.length === 0) { const why = [g, c].map((x) => x.reason && x.reason.message).filter(Boolean).join(" | "); return new Response("Не удалось получить ответ. Причины: " + why, { status: 502, headers: h }); } else if (drafts.length === 1) { answer = drafts[0]; } else { // Склеиваем два ответа в один const question = messages[messages.length - 1].content; const prompt = `Вопрос пользователя:\n${question}\n\n` + `Черновик 1:\n${drafts[0]}\n\nЧерновик 2:\n${drafts[1]}\n\n` + `Объедини черновики в один лучший ответ: сохрани всё верное и полезное, убери повторы, ` + `при расхождениях выбери более точный вариант. Пиши на языке пользователя. ` + `Не упоминай, что были черновики или несколько источников. Выдай только готовый ответ.`; try { answer = await askGemini(env, system, [{ role: "user", content: prompt }]); } catch { answer = drafts[0]; } } return new Response(answer, { headers: { ...h, "Content-Type": "text/plain; charset=utf-8" } }); }, };
